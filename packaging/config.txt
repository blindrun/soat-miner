# SOAT Miner configuration
#
# Edit this file, then run ./soat-miner.sh (Linux) or soat-miner.bat (Windows).
# Anything you set here is passed to the miner. Command-line flags override it.

# ---------------------------------------------------------------- POOL MODE
# Your Ergo payout address. REQUIRED for pool mining.
# Which coin to mine. Change ALGO, POOL and WALLET together.
#
#   Ergo         ALGO=autolykos2  POOL=ergo.herominers.com:1180     WALLET starts with 9
#   Pearl        ALGO=pearl-pow   POOL=pearl.herominers.com:1200    WALLET starts with prl1
#   Bitcoin III  ALGO=sha3-256t   POOL=stratum.pythonpool.dev:3357  WALLET starts with 1, 3 or bc1
#
# Pearl is NVIDIA only: the Vulkan build cannot mine it. Ergo and Bitcoin III
# run on either backend, so AMD and Intel cards can mine both.
ALGO=autolykos2

WALLET=9YOUR_ERGO_ADDRESS_HERE

# Pool to mine to. The reviewed, current profiles are in ergo-pools.json.
# Ready-made Linux/Windows launchers exist for each profile as mine_ergo_*.sh
# and mine_ergo_*.bat. Do not copy a status URL as a Stratum endpoint.
# Default: HeroMiners (ergo.herominers.com:1180).
POOL=ergo.herominers.com:1180

# Worker name shown on the pool dashboard.
WORKER=rig1

# Pool password. Almost always "x".
PASSWORD=x

# --------------------------------------------------------------- LITHOS MODE
# TEST NETWORK ONLY for now: tested end to end against a live Lithos client on
# the Ergo testnet (shares accepted), but not yet validated on mainnet.
# Lithos is a decentralised pool protocol, not a coin or an algorithm - the
# proof of work is the same Autolykos v2. Its reference client runs a stratum
# server on your own machine and settles rewards on-chain, so there is no pool
# operator involved.
#
# Needs a synced Ergo node, Java 11, and the Lithos client running:
#   github.com/Lithos-Protocol/Lithos-Client
#
# Set this to yes to mine into it. POOL below is then ignored, and WALLET is
# not required - payment follows the NODE the Lithos client is attached to, not
# an address set here.
LITHOS=no

# Where the Lithos client's stratum server is listening.
LITHOS_ADDR=127.0.0.1:4444

# ---------------------------------------------------------------- SOLO MODE
# Leave POOL empty above and set these to mine solo against your own node.
# The payout address is configured on the NODE in solo mode, not here.
NODE=127.0.0.1
NODE_PORT=9053

# ------------------------------------------------------------------ TUNING
# Nonces per GPU launch. Larger = slightly better throughput, coarser stop.
BATCH=4194304

# Seconds between readouts.
INTERVAL=5

# Memory clock offset, in MHz of transfer rate.
# CUDA and Vulkan both force performance state P2, which runs your memory
# slower than it is rated for (10251 vs 10501 on a 4090). Autolykos is memory
# bound so that is straight hashrate. 500 puts a 4090 back to stock and is
# worth about 2.5%. This is NOT an overclock at 500.
# Needs root on Linux. 0 leaves the card alone.
MCLK_OFFSET=0

# Build the next block's lookup table ahead of time: auto | on | off
# A new block needs a new 7.27GB table, and building it stops mining for 1.5s
# on a 4090 and 8.2s on a 6700 XT. Building it early removes that.
# It needs two tables in memory, which is 14.5GB, so it does not fit on most
# cards. Auto checks first and turns itself off if it will not fit.
# Worth 0.73% on a 4090. CUDA only for now.
CACHE_DAG=auto

# Backend: auto | cuda | vulkan
#   auto   - use CUDA if an NVIDIA GPU is present, else Vulkan
#   cuda   - NVIDIA only. Faster for Ergo on Ada, and for Bitcoin III on every
#            NVIDIA card measured so far (1543 vs 1086 MH/s on a 4090)
#   vulkan - AMD, Intel, or NVIDIA. The only backend on AMD.
#            Ergo and Bitcoin III both have Vulkan shaders; Pearl does not yet.
#            Measured on an RX 7900 XT: Bitcoin III 540 MH/s at 289 W.
BACKEND=auto
